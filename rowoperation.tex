\begin{itemize}
\item What are \x{row operations} and \x{column operations}?

\end{itemize}

\begin{frame}{Review: Three Views of Matrix Multiplication}
For $C = AB$, we have three interpretations:

\begin{enumerate}
\item \textbf{Column view}: Each column of $C$ is a linear combination of columns of $A$
$$|c_j| = A |b_j|$$

\item \textbf{Row view}: Each row of $C$ is a linear combination of rows of $B$
$$-c_i- = -a_i- B$$

\item \textbf{Sum-of-rank-one}:
$$C = AB = \sum_{k=1}^{n} |a_k| \cdot -b_k-$$
\end{enumerate}
\end{frame}

\begin{frame}{Review: The Ingredient Table Interpretation}
\begin{center}

    node distance=3cm,

\node[box, fill=green!20, right of=A] (B) {Semi-finished\\Products};
\node[box, fill=red!20, right of=B] (C) {Final\\Products};

\draw[-{Stealth[length=3mm]}, thick] (B) -- node[above] {$B$} (C);

\end{tikzpicture}
\end{center}

\vspace{0.3cm}
Matrix $C = AB$ represents the \x{composition} of two recipe tables:
\begin{itemize}
\item $A$: Semi-finished $\leftarrow$ Raw materials
\item $B$: Final products $\leftarrow$ Semi-finished
\item $C = AB$: Final products $\leftarrow$ Raw materials
\end{itemize}
\end{frame}

\begin{frame}{Review: Coffee Shop Example}
\textbf{Matrix $A$} (Semi-finished $\leftarrow$ Raw materials):

$$A = \begin{array}{c|ccc}
 & \text{\milk~Milk} & \text{\coffee~Coffee} & \text{\tea~Tea} \\
\hline
\text{\leaf~Leaf} & 0 & 0 & 2 \\

\vspace{0.3cm}
\small

\begin{itemize}
\item To make \milk: need 1 \cow
\item To make \coffee: need 2 \bean
\item To make \tea: need 2 \leaf~+ 1 \lemon
\end{itemize}
\end{frame}

\begin{frame}{Review: Coffee Shop Example (continued)}
\textbf{Matrix $B$} (Final products $\leftarrow$ Semi-finished):

$$B = \begin{array}{c|cc}
 & \text{\bento~Set 1} & \text{\ramen~Set 2} \\
\hline
\text{\milk~Milk} & 2 & 1 \\
\text{\coffee~Coffee} & 0 & 2 \\
\text{\tea~Tea} & 1 & 1
\end{array}$$

\vspace{0.3cm}
\textbf{Interpretation:}
\begin{itemize}
\item \bento: 2\milk~+ 0\coffee~+ 1\tea
\item \ramen: 1\milk~+ 2\coffee~+ 1\tea
\end{itemize}
\end{frame}

\begin{frame}{Review: Coffee Shop Example (result)}
\textbf{Matrix $C = AB$} (Final products $\leftarrow$ Raw materials):

$$C = \begin{array}{c|cc}
 & \text{\bento~Set 1} & \text{\ramen~Set 2} \\

\text{\leaf~Leaf} & 2 & 2 \\

\begin{itemize}

\item \ramen: needs 2\leaf~+ 1\lemon~+ 4\bean~+ 1\cow
\end{itemize}
\end{frame}

\begin{frame}{Where Each Concept Appears}
\begin{center}
\begin{tabular}{|l|c|c|c|}
\hline
\textbf{Concept} & \textbf{Matrix $A$} & \textbf{Matrix $B$} & \textbf{Matrix $C$} \\
\hline
Raw materials \leaf\lemon\bean\cow & \x{rows} & -- & \x{rows} \\
\hline
Semi-finished \milk\coffee\tea & \x{columns} & \x{rows} & -- \\
\hline

\hline
\end{tabular}
\end{center}

\vspace{0.5cm}
\begin{block}{The Fundamental Principle}
When we \x{redefine} one of these three concepts (change the unit, repackage, bundle), all matrices that reference this concept must update their representation. But the underlying production relationships remain unchanged.
\end{block}
\end{frame}

\section{Row Operations: Contravariant Logic}

\begin{frame}{Row Operations: Redefining Raw Materials}
\textbf{Core Idea:} Row operations correspond to \x{redefining raw materials}.

\vspace{0.5cm}
When we redefine a raw material (change its unit or bundle it with others), we must update:
\begin{itemize}
\item The \x{rows of matrix $A$} (since raw materials appear in rows of $A$)
\item The \x{rows of matrix $C$} (since raw materials appear in rows of $C$)
\item Matrix $B$ is \x{not affected} (semi-finished products don't change)
\end{itemize}

\vspace{0.5cm}
\begin{alertblock}{The Subtle Point}
Row operations have \x{contravariant (inverse) logic} because rows and materials have a \x{multiplication relationship}:
$$\text{(demand coefficient)} \times \text{(material)} = \text{(total amount needed)}$$
\end{alertblock}
\end{frame}

\begin{frame}{Why Contravariant Logic?}
The key equation:
$$\text{demand} \times \text{material} = \text{constant}$$

\vspace{0.3cm}
\textbf{Example with} \bean:
\begin{itemize}
\item Original: need 4 \bean~for \ramen
\item If we redefine: \bean~→ \bean\bean~(double bean, one unit = two old beans)
\item Then we need: 2 \bean\bean~= 4 old \bean~(same total!)
\end{itemize}

\vspace{0.3cm}
\begin{block}{Contravariant = Inverse Direction}
\begin{itemize}
\item Material becomes \x{larger} $\Rightarrow$ Need \x{fewer} units
\item Material becomes \x{smaller} $\Rightarrow$ Need \x{more} units
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Type 1: Row Multiplication}
\begin{defi}[Row Multiplication]
Multiply a row by a nonzero number $k$.

\textbf{Notation:} row $\times k$
\end{defi}

\vspace{0.3cm}
\textbf{What it means:} Change the unit size of a material.

\textbf{Example:} \bean~→ \bean\bean~(double bean)

\vspace{0.3cm}
\begin{alertblock}{Contravariant Logic}
\begin{itemize}
\item Material unit becomes \x{larger} ($\times 2$)
\item Demand numbers become \x{smaller} ($\times \frac{1}{2}$)
\item Row operation: multiply by $\frac{1}{2}$
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{Example 1.1: \bean~→ \bean\bean~(Double Bean)}
\textbf{Original table:}
$$\begin{array}{c|cc}
 & \text{\bento~Set 1} & \text{\ramen~Set 2} \\
\hline
\text{\leaf~Leaf} & 2 & 2 \\
\text{\lemon~Lemon} & 1 & 1 \\
\text{\bean~Bean} & 0 & 4 \\
\text{\cow~Cow} & 2 & 1
\end{array}$$

\vspace{0.3cm}
\textbf{What we're doing:} Redefine \bean~→ \bean\bean

One \bean\bean~= two old \bean. The new unit is \x{larger}!
\end{frame}

\begin{frame}{Example 1.1: How Many Do We Need?}
\textbf{For} \ramen~\textbf{Set 2, originally:}
$$2\text{\leaf} + 1\text{\lemon} + 4\text{\bean} + 1\text{\cow}$$

\textbf{After redefining} \bean~→ \bean\bean:

\vspace{0.2cm}
\begin{itemize}
\item Still need: 2\leaf, 1\lemon, 1\cow~(unchanged)
\item For beans: originally 4\bean
\item How many \bean\bean? Since 1\bean\bean~= 2\bean
\item Need: $\frac{4\bean}{2} = 2$\bean\bean
\end{itemize}

\vspace{0.2cm}
New recipe: $2\text{\leaf} + 1\text{\lemon} + 2\text{\bean\bean} + 1\text{\cow}$

\x{Number changed from 4 → 2!}
\end{frame}

\begin{frame}{Example 1.1: Result}
\textbf{Updated table} (row 3 \x{divided by 2}):
$$\begin{array}{c|cc}
 & \text{\bento~Set 1} & \text{\ramen~Set 2} \\
\hline
\text{\leaf~Leaf} & 2 & 2 \\
\text{\lemon~Lemon} & 1 & 1 \\
\text{\bean\bean~Double bean} & 0 & 2 \\
\text{\cow~Cow} & 2 & 1
\end{array}$$

\textbf{Matrix notation:} Perform $r_3 \to \frac{1}{2}r_3$:
$$\begin{pmatrix}
2 & 2 \\
1 & 1 \\
0 & 4 \\
2 & 1
\end{pmatrix} \xrightarrow{r_3 \to \frac{1}{2}r_3} \begin{pmatrix}
2 & 2 \\
1 & 1 \\
0 & 2 \\
2 & 1
\end{pmatrix}$$

\begin{alertblock}{Key}
Material $\times 2$ → Row $\times \frac{1}{2}$ (contravariant!)
\end{alertblock}
\end{frame}

\begin{frame}{Type 2: Row Switching}
\begin{defi}[Row Switching]
Swap row $i$ and row $j$.

\textbf{Notation:} $r_i \leftrightarrow r_j$
\end{defi}

\vspace{0.3cm}
\textbf{Interpretation:} Reordering materials in the table.

\vspace{0.3cm}
This doesn't change any production relationships, just the order.

\vspace{0.3cm}
\textbf{Example:}
$$\begin{pmatrix}
2 & 2 \\
1 & 1 \\
0 & 4 \\
2 & 1
\end{pmatrix} \xrightarrow{r_1 \leftrightarrow r_2} \begin{pmatrix}
1 & 1 \\
2 & 2 \\
0 & 4 \\
2 & 1
\end{pmatrix}$$
\end{frame}

\begin{frame}{Type 3: Row Addition}
\begin{defi}[Row Addition]
Add a multiple of one row to another row.

\textbf{Notation:} row $i$ + $k \times$ row $j$
\end{defi}

\vspace{0.3cm}
\textbf{What it means:} Bundle materials together.

\textbf{Example:} \lemon~→ \lemon~+ 2\leaf~(lemon bundled with 2 leaves)

\vspace{0.3cm}
\begin{alertblock}{Contravariant Logic - The Tricky Part!}
\begin{itemize}
\item New \lemon~\x{includes} 2\leaf~(material change: +2\leaf)
\item So we need \x{fewer} separate \leaf~(row change: -2\lemon)
\item \x{Opposite signs!} Material +, Row -
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{Example 1.3: \lemon~→ \lemon~+ 2\leaf}
\textbf{Original table:}
$$\begin{array}{c|cc}
 & \text{\bento~Set 1} & \text{\ramen~Set 2} \\
\hline
\text{\leaf~Leaf} & 2 & 2 \\
\text{\lemon~Lemon} & 1 & 1 \\
\text{\bean~Bean} & 0 & 4 \\
\text{\cow~Cow} & 2 & 1
\end{array}$$

\vspace{0.3cm}
\textbf{What we're doing:} Bundle \lemon~with 2\leaf

Now each \lemon~comes with 2\leaf~included (like a combo pack!)
\end{frame}

\begin{frame}{Example 1.3: How Many Do We Need?}
\textbf{For} \bento~\textbf{Set 1, originally:}
$$2\text{\leaf} + 1\text{\lemon} + 0\text{\bean} + 2\text{\cow}$$

\textbf{After bundling} \lemon~→ \lemon~+ 2\leaf:

\vspace{0.2cm}
\begin{itemize}
\item Still need: 1\lemon~(bundled), 0\bean, 2\cow
\item The 1 bundled \lemon~\x{includes} 2\leaf
\item Originally needed: 2\leaf~total
\item \x{Already have} 2\leaf~from the bundled \lemon!
\item Need \x{0 extra} \leaf
\end{itemize}

\vspace{0.2cm}
New recipe: $0\text{\leaf} + 1(\text{\lemon}+2\text{\leaf}) + 0\text{\bean} + 2\text{\cow}$

\x{Leaf number changed from 2 → 0!}
\end{frame}

\begin{frame}{Example 1.3: Result}
\textbf{Updated table} (row operation: $r_1 \to r_1 - 2r_2$):
$$\begin{array}{c|cc}
 & \text{\bento~Set 1} & \text{\ramen~Set 2} \\
\hline
\text{\leaf~Leaf} & 0 & 0 \\
\text{\lemon+2\leaf} & 1 & 1 \\
\text{\bean~Bean} & 0 & 4 \\
\text{\cow~Cow} & 2 & 1
\end{array}$$

\vspace{0.3cm}
\begin{alertblock}{Key - Contravariant!}
\lemon~includes 2\leaf~→ need \x{fewer} separate \leaf~(subtract!)
\end{alertblock}
\end{frame}

\begin{frame}{Summary: Contravariant Logic for Rows}
\begin{center}
\begin{tabular}{|l|c|}
\hline
\textbf{Material Change} & \textbf{Row Change} \\
\hline
\bean~→ \bean\bean~(larger) & row $\times \frac{1}{2}$ (smaller) \\
\hline
\lemon~→ \lemon+2\leaf~(includes \leaf) & \leaf~row - 2×\lemon~row \\
\hline
\end{tabular}
\end{center}

\vspace{0.5cm}
\begin{alertblock}{Remember}
Rows work \x{backwards} from the material change!
\end{alertblock}
\end{frame}

\section{Column Operations: Covariant Logic}

\begin{frame}{Column Operations: Redefining Products}
\textbf{Core Idea:} Column operations correspond to \x{redefining final products}.

\vspace{0.5cm}
When we redefine a final product (change its portion size or bundle it), we must update:
\begin{itemize}
\item The \x{columns of matrix $B$} (since products appear in columns of $B$)
\item The \x{columns of matrix $C$} (since products appear in columns of $C$)
\item Matrix $A$ is \x{not affected} (raw materials don't change)
\end{itemize}

\vspace{0.5cm}
\begin{alertblock}{The Crucial Difference}
Column operations have \x{covariant (direct) logic} because columns and products have an \x{equation relationship}:
$$\text{(product)} = \text{(recipe)}$$
They appear on opposite sides of the equals sign!
\end{alertblock}
\end{frame}

\begin{frame}{Why Covariant Logic?}
The key equation:
$$\text{product} = \text{recipe}$$

\vspace{0.3cm}
\textbf{Example with} \bento:
\begin{itemize}
\item Original \bento: needs 2\leaf~+ 1\lemon~+ 2\cow
\item If we make: 2×\bento~(double portion)
\item Then need: $2 \times (2\text{\leaf} + 1\text{\lemon} + 2\text{\cow})$
\item = 4\leaf~+ 2\lemon~+ 4\cow~(\x{everything doubles!})
\end{itemize}

\vspace{0.3cm}
\begin{block}{Covariant = Same Direction}
\begin{itemize}
\item Product becomes \x{larger} $\Rightarrow$ Recipe becomes \x{larger}
\item Product becomes \x{smaller} $\Rightarrow$ Recipe becomes \x{smaller}
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Type 1: Column Multiplication}
\begin{defi}[Column Multiplication]
Multiply a column by a nonzero number $k$.

\textbf{Notation:} column $\times k$
\end{defi}

\vspace{0.3cm}
\textbf{What it means:} Make a bigger or smaller portion.

\textbf{Example:} \bento~→ 2×\bento~(double portion)

\vspace{0.3cm}
\begin{alertblock}{Covariant Logic}
\begin{itemize}
\item Product becomes \x{larger} ($\times 2$)
\item Recipe numbers become \x{larger} ($\times 2$)
\item Column operation: multiply by $2$
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{Example 1.4: \bento~→ 2×\bento}
\textbf{Original table:}
$$\begin{array}{c|cc}
 & \text{\bento~Set 1} & \text{\ramen~Set 2} \\
\hline
\text{\leaf~Leaf} & 2 & 2 \\
\text{\lemon~Lemon} & 1 & 1 \\
\text{\bean~Bean} & 0 & 4 \\
\text{\cow~Cow} & 2 & 1
\end{array}$$

\vspace{0.3cm}
\textbf{What we're doing:} Make 2×\bento~(double portion)

\textbf{What happens?}
\begin{itemize}
\item Original \bento: 2\leaf~+ 1\lemon~+ 0\bean~+ 2\cow
\item Double \bento: $2 \times$ everything = 4\leaf~+ 2\lemon~+ 0\bean~+ 4\cow
\end{itemize}
\end{frame}

\begin{frame}{Example 1.4: Result}
\textbf{Updated table} (column 1 \x{multiplied by 2}):
$$\begin{array}{c|cc}
 & \text{\bento\bento~Double Set 1} & \text{\ramen~Set 2} \\
\hline
\text{\leaf~Leaf} & 4 & 2 \\
\text{\lemon~Lemon} & 2 & 1 \\
\text{\bean~Bean} & 0 & 4 \\
\text{\cow~Cow} & 4 & 1
\end{array}$$

\textbf{Matrix notation:} Perform $c_1 \to 2c_1$:
$$\begin{pmatrix}
2 & 2 \\
1 & 1 \\
0 & 4 \\
2 & 1
\end{pmatrix} \xrightarrow{c_1 \to 2c_1} \begin{pmatrix}
4 & 2 \\
2 & 1 \\
0 & 4 \\
4 & 1
\end{pmatrix}$$

\begin{alertblock}{Key}
Product $\times 2$ → Column $\times 2$ (\x{same direction!})
\end{alertblock}
\end{frame}

\begin{frame}{Type 2: Column Switching}
\begin{defi}[Column Switching]
Swap column $i$ and column $j$.

\textbf{Notation:} $c_i \leftrightarrow c_j$
\end{defi}

\vspace{0.3cm}
\textbf{Interpretation:} Reordering products in the table.

\vspace{0.3cm}
\textbf{Example:}
$$\begin{pmatrix}
2 & 2 \\
1 & 1 \\
0 & 4 \\
2 & 1
\end{pmatrix} \xrightarrow{c_1 \leftrightarrow c_2} \begin{pmatrix}
2 & 2 \\
1 & 1 \\
4 & 0 \\
1 & 2
\end{pmatrix}$$
\end{frame}

\begin{frame}{Type 3: Column Addition}
\begin{defi}[Column Addition]
Add a multiple of one column to another column.

\textbf{Notation:} column $j$ + $k \times$ column $i$
\end{defi}

\vspace{0.3cm}
\textbf{What it means:} Bundle products together.

\textbf{Example:} \ramen~→ \ramen~+ \bento~(combo meal!)

\vspace{0.3cm}
\begin{alertblock}{Covariant Logic - Straightforward!}
\begin{itemize}
\item New \ramen~\x{includes} \bento~(product change: +\bento)
\item Recipe needs ingredients for \x{both} (recipe change: +\bento~recipe)
\item \x{Same signs!} Product +, Recipe +
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{Example 1.6: \ramen~→ \ramen~+ \bento}
\textbf{Original table:}
$$\begin{array}{c|cc}
 & \text{\bento~Set 1} & \text{\ramen~Set 2} \\
\hline
\text{\leaf~Leaf} & 2 & 2 \\
\text{\lemon~Lemon} & 1 & 1 \\
\text{\bean~Bean} & 0 & 4 \\
\text{\cow~Cow} & 2 & 1
\end{array}$$

\vspace{0.3cm}
\textbf{What we're doing:} Bundle \ramen~+ \bento~(combo!)

\textbf{Original recipes:}
\begin{itemize}
\item \bento: 2\leaf~+ 1\lemon~+ 0\bean~+ 2\cow
\item \ramen: 2\leaf~+ 1\lemon~+ 4\bean~+ 1\cow
\end{itemize}
\end{frame}

\begin{frame}{Example 1.6: How Much Do We Need?}
\textbf{For the combo} \ramen~+ \bento:

\vspace{0.2cm}
\begin{itemize}
\item \ramen~needs: 2\leaf~+ 1\lemon~+ 4\bean~+ 1\cow
\item \bento~needs: 2\leaf~+ 1\lemon~+ 0\bean~+ 2\cow
\item \x{Add them together:}
  \begin{itemize}
  \item Leaf: $2 + 2 = 4$\leaf
  \item Lemon: $1 + 1 = 2$\lemon
  \item Bean: $4 + 0 = 4$\bean
  \item Cow: $1 + 2 = 3$\cow
  \end{itemize}
\end{itemize}

\vspace{0.2cm}
New recipe: 4\leaf~+ 2\lemon~+ 4\bean~+ 3\cow

\x{All numbers increased!} (covariant)
\end{frame}

\begin{frame}{Example 1.6: Result}
\textbf{Updated table} (column 2: $c_2 \to c_2 + c_1$, \x{add not subtract}):
$$\begin{array}{c|cc}
 & \text{\bento~Set 1} & \text{\ramen\bento~Combo} \\
\hline
\text{\leaf~Leaf} & 2 & 4 \\
\text{\lemon~Lemon} & 1 & 2 \\
\text{\bean~Bean} & 0 & 4 \\
\text{\cow~Cow} & 2 & 3
\end{array}$$

\textbf{Matrix notation:} Perform $c_2 \to c_2 + c_1$:
$$\begin{pmatrix}
2 & 2 \\
1 & 1 \\
0 & 4 \\
2 & 1
\end{pmatrix} \xrightarrow{c_2 \to c_2 + c_1} \begin{pmatrix}
2 & 4 \\
1 & 2 \\
0 & 4 \\
2 & 3
\end{pmatrix}$$

\begin{alertblock}{Key}
Product becomes $(\text{\ramen} + \text{\bento})$ → Recipe \x{adds} the two recipes!
\end{alertblock}
\end{frame}

\begin{frame}{Summary: Covariant Logic for Columns}
\begin{center}
\begin{tabular}{|l|c|}
\hline
\textbf{Product Change} & \textbf{Column Change} \\
\hline
\bento~→ 2×\bento~(larger) & column $\times 2$ (larger) \\
\hline
\ramen~→ \ramen+\bento~(includes \bento) & \ramen~col + \bento~col \\
\hline
\end{tabular}
\end{center}

\vspace{0.5cm}
\begin{alertblock}{Remember}
Columns work \x{forwards} with the product change!
\end{alertblock}
\end{frame}

\begin{frame}{THE KEY DISTINCTION}
\begin{center}
\begin{tabular}{|l|c|c|}
\hline
 & \textbf{Rows} & \textbf{Columns} \\
\hline
\textbf{Logic} & Contravariant & Covariant \\
 & (inverse) & (direct) \\
\hline
\textbf{Example} & \bean~→ \bean\bean & \bento~→ 2×\bento \\
\hline
\textbf{Change} & material $\times 2$ & product $\times 2$ \\
 & row $\times \frac{1}{2}$ & column $\times 2$ \\
\hline
\end{tabular}
\end{center}

\vspace{0.5cm}
\begin{alertblock}{The Essence}
\Large \textbf{Rows work backwards. Columns work forwards.}
\end{alertblock}
\end{frame}

\begin{frame}{Why We Don't Update Semi-finished Products}
Semi-finished products (\milk, \coffee, \tea) appear in:
\begin{itemize}
\item \x{Columns of $A$} (as outputs) → would need column operations (covariant)
\item \x{Rows of $B$} (as inputs) → would need row operations (contravariant)
\end{itemize}

\vspace{0.5cm}
\begin{alertblock}{The Problem}
Updating semi-finished products would require:
\begin{itemize}
\item \x{Two different types of operations}
\item With \x{opposite logic} (contravariant vs covariant)
\item On \x{two different matrices} (both $A$ and $B$)
\end{itemize}
\end{alertblock}

\vspace{0.3cm}
This is significantly more complex than row-only or column-only operations.

\vspace{0.3cm}
\textbf{Solution:} We avoid this complication and focus on raw materials (rows) and final products (columns).
\end{frame}

% §3 SIMULTANEOUS OPERATIONS PRESERVE EQUATIONS
% ═══════════════════════════════════════════════════════════════

\section{Simultaneous Operations Preserve Equations}

\begin{frame}{Change of Perspective}
\textbf{What we've learned so far:}
\begin{itemize}
\item \x{Why} rows are contravariant (multiplication relationship)
\item \x{Why} columns are covariant (equation relationship)
\item \x{How} to update tables when redefining materials/products
\end{itemize}

\vspace{0.5cm}
\begin{alertblock}{From This Point Forward}
We \x{forget about materials/products} and focus only on:
\begin{itemize}
\item \textbf{What}: Matrices $A$, $B$, $C$
\item \textbf{How}: Simultaneous operations keep equations valid
\end{itemize}
\end{alertblock}

\vspace{0.3cm}
This simplifies our work: we only track \x{matrix changes}, not material redefinitions.
\end{frame}

\begin{frame}{Proposition 1.1: Row Operation Invariance}
\begin{prop}[Row Operation Invariance]
In the equation $C = AB$, if we perform the \x{same row operation} on both $A$ and $C$ simultaneously, the equality remains valid.
\end{prop}

\vspace{0.3cm}
\textbf{Why?} Raw materials appear in the \x{rows} of both $A$ and $C$. When we redefine raw materials, both must update. Matrix $B$ doesn't reference raw materials, so it doesn't change.

\vspace{0.3cm}
\textbf{But from now on}, we don't need to think about "why" — we just use the fact that \x{simultaneous row operations on $A$ and $C$ preserve the equation}.
\end{frame}

\begin{frame}{Example 1.7: Row Operation Preserves Equation}
Consider:
$$\underbrace{\begin{pmatrix} 2 & 0 \\ 0 & 1 \end{pmatrix}}_{A} \underbrace{\begin{pmatrix} 1 & 1 \\ 2 & 3 \end{pmatrix}}_{B} = \underbrace{\begin{pmatrix} 2 & 2 \\ 2 & 3 \end{pmatrix}}_{C}$$

Perform $r_1 \to 2r_1$ on \x{both $A$ and $C$}:
$$\underbrace{\begin{pmatrix} 4 & 0 \\ 0 & 1 \end{pmatrix}}_{A'} \underbrace{\begin{pmatrix} 1 & 1 \\ 2 & 3 \end{pmatrix}}_{B} = \underbrace{\begin{pmatrix} 4 & 4 \\ 2 & 3 \end{pmatrix}}_{C'}$$

\begin{center}
\Large \textbf{Still valid!} $\checkmark$
\end{center}

\vspace{0.3cm}
\textbf{Note:} $B$ is unchanged because it doesn't have raw materials in its rows.
\end{frame}

\begin{frame}{Proposition 1.2: Column Operation Invariance}
\begin{prop}[Column Operation Invariance]
In the equation $C = AB$, if we perform the \x{same column operation} on both $B$ and $C$ simultaneously, the equality remains valid.
\end{prop}

\vspace{0.3cm}
\textbf{Why?} Final products appear in the \x{columns} of both $B$ and $C$. When we redefine final products, both must update. Matrix $A$ doesn't reference final products, so it doesn't change.

\vspace{0.3cm}
\textbf{But from now on}, we don't need to think about products — we just use the fact that \x{simultaneous column operations on $B$ and $C$ preserve the equation}.
\end{frame}

\begin{frame}{Example 1.8: Column Operation Preserves Equation}
Consider:
$$\underbrace{\begin{pmatrix} 1 & 0 \\ 0 & 2 \end{pmatrix}}_{A} \underbrace{\begin{pmatrix} 2 & 1 \\ 1 & 3 \end{pmatrix}}_{B} = \underbrace{\begin{pmatrix} 2 & 1 \\ 2 & 6 \end{pmatrix}}_{C}$$

Perform $c_1 \to 3c_1$ on \x{both $B$ and $C$}:
$$\underbrace{\begin{pmatrix} 1 & 0 \\ 0 & 2 \end{pmatrix}}_{A} \underbrace{\begin{pmatrix} 6 & 1 \\ 3 & 3 \end{pmatrix}}_{B'} = \underbrace{\begin{pmatrix} 6 & 1 \\ 6 & 6 \end{pmatrix}}_{C'}$$

\begin{center}
\Large \textbf{Still valid!} $\checkmark$
\end{center}

\vspace{0.3cm}
\textbf{Note:} $A$ is unchanged because it doesn't have final products in its columns.
\end{frame}

\begin{frame}{Why These Propositions Matter}
\begin{block}{Foundation for Matrix Equations}
Propositions 1.1 and 1.2 provide the theoretical basis for:
\begin{itemize}
\item Solving matrix equations $AB = C$
\item Finding matrix inverses $A^{-1}$
\item Understanding how operations affect equations
\end{itemize}
\end{block}

\vspace{0.5cm}
\textbf{Key insight:} We can \x{manipulate} equations $C = AB$ using row/column operations, just like we manipulate algebraic equations, as long as we:
\begin{itemize}
\item Apply row operations to \x{both $A$ and $C$}
\item Apply column operations to \x{both $B$ and $C$}
\end{itemize}
\end{frame}

\section{Matrix Inverses}

\begin{frame}{Definition: Matrix Inverse}
\begin{defi}[Matrix Inverse]
An $n \times n$ matrix $A$ is called \x{invertible} if there exists an $n \times n$ matrix $B$ such that
$$AB = BA = I_n$$
where $I_n$ is the $n \times n$ identity matrix. We call $B$ the \x{inverse} of $A$ and denote it $A^{-1}$.
\end{defi}

\vspace{0.5cm}
\textbf{Interpretation:} If $A$ is the recipe "raw materials $\to$ products", then $A^{-1}$ is the recipe "products $\to$ raw materials".
\end{frame}

\begin{frame}{Example 1.9: Coffee Shop Inverse}
Consider:
$$A = \begin{pmatrix} 0 & 2 \\ 1 & 0 \end{pmatrix} = \begin{array}{c|cc}
& \text{\coffee~Coffee} & \text{\milk~Milk} \\
\hline
\text{\bean~Bean} & 0 & 2 \\
\text{\cow~Cow} & 1 & 0
\end{array}$$

Then:
$$A^{-1} = \begin{pmatrix} 0 & 1 \\ \frac{1}{2} & 0 \end{pmatrix} = \begin{array}{c|cc}
& \text{\bean~Bean} & \text{\cow~Cow} \\
\hline
\text{\coffee~Coffee} & 0 & 1 \\
\text{\milk~Milk} & \frac{1}{2} & 0
\end{array}$$

\textbf{Verification:}
$$AA^{-1} = \begin{pmatrix} 0 & 2 \\ 1 & 0 \end{pmatrix} \begin{pmatrix} 0 & 1 \\ \frac{1}{2} & 0 \end{pmatrix} = \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix} = I$$
\end{frame}

\begin{frame}{Proposition 1.3: Inverse is Unique}
\begin{prop}[Uniqueness of Inverse]
If $A$ has an inverse, it is unique.
\end{prop}

\vspace{0.3cm}
\textbf{Proof:} Suppose $B$ and $C$ both satisfy the definition. Then:
$$B = BI = B(AC) = (BA)C = IC = C$$
Therefore $B = C$. \qed
\end{frame}

\begin{frame}{Proposition 1.4: Product of Invertible Matrices}
\begin{prop}[Inverse of Product]
If $A$ and $B$ are invertible, then $AB$ is invertible and
$$(AB)^{-1} = B^{-1}A^{-1}$$
\end{prop}

\vspace{0.3cm}
\textbf{Verification:}
$$(AB)(B^{-1}A^{-1}) = A(BB^{-1})A^{-1} = AIA^{-1} = AA^{-1} = I$$

\vspace{0.5cm}
\begin{alertblock}{Order Reversal}
$(AB)^{-1} = B^{-1}A^{-1}$, \x{not} $A^{-1}B^{-1}$.

\textbf{Intuition:} Socks then shoes $\to$ undo by removing shoes first, then socks.
\end{alertblock}
\end{frame}

% §5 WHY INTRODUCE INVERSES: THE TWO CRITICAL THEOREMS
% ═══════════════════════════════════════════════════════════════

\section{Why Introduce Inverses?}

\begin{frame}{Why Do We Need Matrix Inverses?}
\textbf{Question:} Why introduce the concept of inverse matrices?

\vspace{0.5cm}
\textbf{Answer:} Inverses provide \x{notational convenience} for tracking invariants when solving matrix equations.

\vspace{0.5cm}
\begin{block}{The Key Insight}
Writing expressions like $B = A^{-1}C$ or $B = CA^{-1}$ makes explicit that:
\begin{itemize}
\item "\x{$B$ doesn't change}" when we perform operations
\item We can \x{simultaneously transform} the "$A^{-1}$" part and the "$C$" part
\item The transformation preserves the equation
\end{itemize}
\end{block}

\vspace{0.3cm}
The next two theorems are the \x{most important} results for practical computation!
\end{frame}

\begin{frame}{Theorem 1.6: Row Operations on $A^{-1}C$}
\begin{prop}[Row Operations Preserve $A^{-1}C$]
In the expression $B = A^{-1}C$, if we perform the \x{same row operation} on both "$A$" and "$C$" simultaneously, we get:
$$B = (A')^{-1}C'$$
where $A'$, $C'$ are the matrices after the row operation, and \x{$B$ remains unchanged}.
\end{prop}

\vspace{0.3cm}
\textbf{Why?} From $AB = C$, applying row operations to both $A$ and $C$ preserves the equation (Prop 1.1):
$$A'B = C'$$
Therefore: $B = (A')^{-1}C'$

\vspace{0.3cm}
\begin{alertblock}{Key Application}
This theorem allows us to solve $AB = C$ for $B$ by performing row operations!
\end{alertblock}
\end{frame}

\begin{frame}{Example 1.10: Using $A^{-1}C$ Row Operations (Setup)}
\textbf{Problem:} Solve $AB = C$ for $B$ where
$$A = \begin{pmatrix} 2 & 1 \\ 1 & 1 \end{pmatrix}, \quad C = \begin{pmatrix} 5 & 8 \\ 3 & 5 \end{pmatrix}$$

\textbf{Method:} Write $B = A^{-1}C$ and perform row operations to transform "$A$" into $I$:

$$B = \begin{pmatrix} 2 & 1 \\ 1 & 1 \end{pmatrix}^{-1} \begin{pmatrix} 5 & 8 \\ 3 & 5 \end{pmatrix}$$

\textbf{Goal:} Transform "$A$" to "$I$" using row operations.
\end{frame}

\begin{frame}{Example 1.10: Step 1-2}
\textbf{Step 1:} $r_1 \leftrightarrow r_2$
$$B = \begin{pmatrix} 1 & 1 \\ 2 & 1 \end{pmatrix}^{-1} \begin{pmatrix} 3 & 5 \\ 5 & 8 \end{pmatrix}$$

\textbf{Step 2:} $r_2 \to r_2 - 2r_1$
$$B = \begin{pmatrix} 1 & 1 \\ 0 & -1 \end{pmatrix}^{-1} \begin{pmatrix} 3 & 5 \\ -1 & -2 \end{pmatrix}$$
\end{frame}

\begin{frame}{Example 1.10: Step 3-4}
\textbf{Step 3:} $r_2 \to (-1) \cdot r_2$
$$B = \begin{pmatrix} 1 & 1 \\ 0 & 1 \end{pmatrix}^{-1} \begin{pmatrix} 3 & 5 \\ 1 & 2 \end{pmatrix}$$

\textbf{Step 4:} $r_1 \to r_1 - r_2$
$$B = \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}^{-1} \begin{pmatrix} 2 & 3 \\ 1 & 2 \end{pmatrix} = I^{-1} \begin{pmatrix} 2 & 3 \\ 1 & 2 \end{pmatrix}$$

Since $I^{-1} = I$:
$$B = \begin{pmatrix} 2 & 3 \\ 1 & 2 \end{pmatrix}$$
\end{frame}

\begin{frame}{Example 1.10: Verification}
\textbf{Check:} Does $AB = C$?
$$\begin{pmatrix} 2 & 1 \\ 1 & 1 \end{pmatrix} \begin{pmatrix} 2 & 3 \\ 1 & 2 \end{pmatrix} = \begin{pmatrix} 4+1 & 6+2 \\ 2+1 & 3+2 \end{pmatrix} = \begin{pmatrix} 5 & 8 \\ 3 & 5 \end{pmatrix}$$

\begin{center}
\Large \textbf{Yes!} $\checkmark$
\end{center}

\vspace{0.3cm}
\begin{alertblock}{Key Observation}
We transformed "$A$" to "$I$" using row operations, and automatically obtained $B$!
\end{alertblock}
\end{frame}

\begin{frame}{Theorem 1.7: Column Operations on $CB^{-1}$}
\begin{prop}[Column Operations Preserve $CB^{-1}$]
In the expression $A = CB^{-1}$, if we perform the \x{same column operation} on both "$C$" and "$B$" simultaneously, we get:
$$A = C'(B')^{-1}$$
where $C'$, $B'$ are the matrices after the column operation, and \x{$A$ remains unchanged}.
\end{prop}

\vspace{0.3cm}
\textbf{Why?} From $AB = C$, applying column operations to both $B$ and $C$ preserves the equation (Prop 1.2):
$$AB' = C'$$
Therefore: $A = C'(B')^{-1}$

\vspace{0.3cm}
\begin{alertblock}{Key Application}
This theorem allows us to solve $AB = C$ for $A$ by performing column operations!
\end{alertblock}
\end{frame}

\begin{frame}{Example 1.11: Using $CB^{-1}$ Column Operations (Setup)}
\textbf{Problem:} Solve $AB = C$ for $A$ where
$$B = \begin{pmatrix} 2 & 1 \\ 1 & 1 \end{pmatrix}, \quad C = \begin{pmatrix} 5 & 3 \\ 8 & 5 \end{pmatrix}$$

\textbf{Method:} Write $A = CB^{-1}$ and perform column operations to transform "$B$" into $I$:

$$A = \begin{pmatrix} 5 & 3 \\ 8 & 5 \end{pmatrix} \begin{pmatrix} 2 & 1 \\ 1 & 1 \end{pmatrix}^{-1}$$

\textbf{Goal:} Transform "$B$" to "$I$" using column operations.
\end{frame}

\begin{frame}{Example 1.11: Step 1-2}
\textbf{Step 1:} $c_1 \leftrightarrow c_2$
$$A = \begin{pmatrix} 3 & 5 \\ 5 & 8 \end{pmatrix} \begin{pmatrix} 1 & 2 \\ 1 & 1 \end{pmatrix}^{-1}$$

\textbf{Step 2:} $c_2 \to c_2 - 2c_1$
$$A = \begin{pmatrix} 3 & -1 \\ 5 & -2 \end{pmatrix} \begin{pmatrix} 1 & 0 \\ 1 & -1 \end{pmatrix}^{-1}$$
\end{frame}

\begin{frame}{Example 1.11: Step 3-4}
\textbf{Step 3:} $c_2 \to (-1) \cdot c_2$
$$A = \begin{pmatrix} 3 & 1 \\ 5 & 2 \end{pmatrix} \begin{pmatrix} 1 & 0 \\ 1 & 1 \end{pmatrix}^{-1}$$

\textbf{Step 4:} $c_1 \to c_1 - c_2$
$$A = \begin{pmatrix} 2 & 1 \\ 3 & 2 \end{pmatrix} \begin{pmatrix} 0 & 0 \\ 1 & 1 \end{pmatrix}^{-1} = \begin{pmatrix} 2 & 1 \\ 3 & 2 \end{pmatrix} I^{-1}$$

Wait, error! Let me recalculate...

Actually:
$$A = \begin{pmatrix} 2 & 1 \\ 3 & 2 \end{pmatrix} \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}^{-1} = \begin{pmatrix} 2 & 1 \\ 3 & 2 \end{pmatrix}$$
\end{frame}

\begin{frame}{Summary: The Two Critical Theorems}
\begin{block}{Theorem 1.6: Row Operations on $A^{-1}C$}
Simultaneous row operations on "$A$" and "$C$" in $B = A^{-1}C$ preserve $B$.

\textbf{Use case:} Solving $AB = C$ for $B$
\end{block}

\vspace{0.3cm}
\begin{block}{Theorem 1.7: Column Operations on $CB^{-1}$}
Simultaneous column operations on "$C$" and "$B$" in $A = CB^{-1}$ preserve $A$.

\textbf{Use case:} Solving $AB = C$ for $A$
\end{block}

\vspace{0.5cm}
\begin{alertblock}{Why These Matter}
These two theorems are the \x{foundation} for:
\begin{itemize}
\item Finding matrix inverses
\item Solving matrix equations
\item Understanding matrix transformations
\end{itemize}
\end{alertblock}
\end{frame}

\section{Finding Matrix Inverses}

\begin{frame}{Finding $A^{-1}$: The Method}
\textbf{Goal:} Find $A^{-1}$ for a given matrix $A$.

\vspace{0.3cm}
\textbf{Idea:} Solving $AA^{-1} = I$ for $A^{-1}$ is a special case of solving $AB = C$.

\vspace{0.3cm}
From $A^{-1} = A^{-1}I$, perform \x{row operations} to transform "$A$" into $I$:
$$A^{-1} = A^{-1}I \xrightarrow{\text{row ops}} I^{-1} \cdot (\text{result}) = \text{result}$$

\vspace{0.3cm}
\textbf{Algorithm:}
\begin{enumerate}
\item Write $A^{-1} = A^{-1}I$
\item Apply row operations to both "$A$" and "$I$" until "$A$" becomes $I$
\item The result on the right is $A^{-1}$
\end{enumerate}
\end{frame}

\begin{frame}{Example 1.12: Finding Inverse (Row Operations)}
\textbf{Find:} $A^{-1}$ where $A = \begin{pmatrix} 2 & 1 \\ 3 & 2 \end{pmatrix}$

\textbf{Setup:}
$$A^{-1} = \begin{pmatrix} 2 & 1 \\ 3 & 2 \end{pmatrix}^{-1} \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}$$

\textbf{Step 1:} $r_2 \to r_2 - r_1$
$$A^{-1} = \begin{pmatrix} 2 & 1 \\ 1 & 1 \end{pmatrix}^{-1} \begin{pmatrix} 1 & 0 \\ -1 & 1 \end{pmatrix}$$
\end{frame}

\begin{frame}{Example 1.12: Steps 2-5}
\textbf{Step 2:} $r_1 \leftrightarrow r_2$
$$A^{-1} = \begin{pmatrix} 1 & 1 \\ 2 & 1 \end{pmatrix}^{-1} \begin{pmatrix} -1 & 1 \\ 1 & 0 \end{pmatrix}$$

\textbf{Step 3:} $r_2 \to r_2 - 2r_1$
$$A^{-1} = \begin{pmatrix} 1 & 1 \\ 0 & -1 \end{pmatrix}^{-1} \begin{pmatrix} -1 & 1 \\ 3 & -2 \end{pmatrix}$$

\textbf{Step 4:} $r_2 \to (-1) \cdot r_2$
$$A^{-1} = \begin{pmatrix} 1 & 1 \\ 0 & 1 \end{pmatrix}^{-1} \begin{pmatrix} -1 & 1 \\ -3 & 2 \end{pmatrix}$$

\textbf{Step 5:} $r_1 \to r_1 - r_2$
$$A^{-1} = \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}^{-1} \begin{pmatrix} 2 & -1 \\ -3 & 2 \end{pmatrix} = \begin{pmatrix} 2 & -1 \\ -3 & 2 \end{pmatrix}$$
\end{frame}

\begin{frame}{Example 1.13: Finding Inverse (Column Operations)}
\textbf{Alternative:} Use column operations on $IA^{-1} = A^{-1}$

\textbf{Setup:}
$$A^{-1} = \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix} \begin{pmatrix} 2 & 1 \\ 3 & 2 \end{pmatrix}^{-1}$$

\textbf{Step 1:} $c_1 \to c_1 - c_2$
$$A^{-1} = \begin{pmatrix} 1 & 0 \\ -1 & 1 \end{pmatrix} \begin{pmatrix} 1 & 1 \\ 1 & 2 \end{pmatrix}^{-1}$$

\textbf{Step 2:} $c_2 \to c_2 - c_1$
$$A^{-1} = \begin{pmatrix} 1 & -1 \\ -1 & 2 \end{pmatrix} \begin{pmatrix} 1 & 0 \\ 1 & 1 \end{pmatrix}^{-1}$$
\end{frame}

\begin{frame}{Example 1.13: Final Step}
\textbf{Step 3:} $c_1 \to c_1 - c_2$
$$A^{-1} = \begin{pmatrix} 2 & -1 \\ -3 & 2 \end{pmatrix} \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}^{-1} = \begin{pmatrix} 2 & -1 \\ -3 & 2 \end{pmatrix}$$

\vspace{0.5cm}
\begin{center}
\Large \textbf{Same answer!} $\checkmark$
\end{center}

\vspace{0.3cm}
\begin{block}{Observation}
Both row operations and column operations give the same inverse.
\end{block}
\end{frame}

\section{When Does an Inverse Exist?}

\begin{frame}{Not All Matrices Are Invertible}
\textbf{Example:}
$$A = \begin{pmatrix} 2 & 1 \\ 4 & 2 \end{pmatrix}$$

\textbf{Observation:} Row 2 = $2 \times$ Row 1 (redundant information)

\vspace{0.3cm}
\textbf{Try to find inverse:}
$$A^{-1} = \begin{pmatrix} 2 & 1 \\ 4 & 2 \end{pmatrix}^{-1} \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}$$

Perform $r_2 \to r_2 - 2r_1$:
$$A^{-1} = \begin{pmatrix} 2 & 1 \\ 0 & 0 \end{pmatrix}^{-1} \begin{pmatrix} 1 & 0 \\ -2 & 1 \end{pmatrix}$$

\begin{alertblock}{Stuck!}
Cannot transform "$A$" to $I$. This matrix has \x{no inverse}.
\end{alertblock}
\end{frame}

\begin{frame}{Invertibility Criterion (Preview)}
\begin{block}{Criterion}
An $n \times n$ matrix is invertible if and only if it can be transformed to $I$ using row operations.

Equivalently: \x{rank} = $n$.
\end{block}

\vspace{0.5cm}
\textbf{Detailed discussion} of rank and solvability conditions will be covered in \x{Chapter 2}.

\vspace{0.5cm}
\textbf{For now:}
\begin{itemize}
\item If row operations can reduce $A$ to $I$: invertible
\item If we get a zero row: not invertible
\end{itemize}
\end{frame}

\begin{frame}{Visual Summary: Invertible vs Non-invertible}
\begin{center}
\begin{tabular}{|c|c|}
\hline
\textbf{Invertible} & \textbf{Non-invertible} \\
\hline
$\begin{pmatrix} 2 & 1 \\ 3 & 2 \end{pmatrix}$ & $\begin{pmatrix} 2 & 1 \\ 4 & 2 \end{pmatrix}$ \\
\hline
Can reduce to $I$ & Gets zero row \\
\hline
Rows are independent & Rows are dependent \\
\hline
Full rank & Not full rank \\
\hline
Unique inverse $A^{-1}$ & No inverse \\
\hline
\end{tabular}
\end{center}
\end{frame}

\begin{frame}{Summary: Key Takeaways}
\begin{enumerate}
\item \textbf{Row operations}: \x{Contravariant} (inverse logic)
   \begin{itemize}
   \item Material $\times k$ → Row $\times \frac{1}{k}$ (reciprocal)
   \item Material $x_i \to x_i + kx_j$ → Row $r_j \to r_j - kr_i$ (opposite sign!)
   \end{itemize}

\item \textbf{Column operations}: \x{Covariant} (direct logic)
   \begin{itemize}
   \item Product $\times k$ → Column $\times k$ (same)
   \item Product $y_j \to y_j + ky_i$ → Column $c_j \to c_j + kc_i$ (same sign!)
   \end{itemize}

\item \textbf{Equation $C = AB$ is invariant} under:
   \begin{itemize}
   \item Simultaneous row ops on $A$ and $C$ (Prop 1.1)
   \item Simultaneous column ops on $B$ and $C$ (Prop 1.2)
   \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Summary: Critical Theorems}
\begin{enumerate}
\setcounter{enumi}{3}
\item \textbf{Left inverse = right inverse} for square matrices (Prop 1.4)

\item \textbf{Row operations on $A^{-1}C$} preserve $B$ (Theorem 1.6)
   \begin{itemize}
   \item Use to solve $AB = C$ for $B$
   \end{itemize}

\item \textbf{Column operations on $CB^{-1}$} preserve $A$ (Theorem 1.7)
   \begin{itemize}
   \item Use to solve $AB = C$ for $A$
   \end{itemize}

\item \textbf{Finding $A^{-1}$}: use row ops on $A^{-1}I$ or column ops on $IA^{-1}$ until "$A$" becomes $I$
\end{enumerate}

\vspace{0.5cm}
\begin{alertblock}{Remember}
\textbf{Rows work backwards. Columns work forwards.}
\end{alertblock}
\end{frame}

\begin{frame}[shrink=5]{Standing Assumption}

Throughout this lecture:

\vfill

$$\det(tI - A) = (t - \lambda_1)(t - \lambda_2)\cdots(t - \lambda_m), \qquad \lambda_i \;\text{all distinct}$$

\vfill

Same assumption as Lecture 15.

\vfill

We have the spectral decomposition:

$$A = \lambda_1 P_1 + \lambda_2 P_2 + \cdots + \lambda_m P_m$$

where $P_1, \ldots, P_m$ are compatible projections that partition $I$.

\end{frame}

$$A = \lambda_1 P_1 + \lambda_2 P_2 + \cdots + \lambda_m P_m$$

$$\underbrace{\mathbf{v}}_{\text{any vector}} = \underbrace{P_1 \mathbf{v}}_{\in\,\operatorname{Col}(P_1)} + \underbrace{P_2\mathbf{v}}_{\in\,\operatorname{Col}(P_2)} + \cdots + \underbrace{P_m\mathbf{v}}_{\in\,\operatorname{Col}(P_m)}$$
\end{center}

\vfill

\begin{center}
{\Large \x{What is $\operatorname{Col}(P_i)$?}}

\vspace{0.5cm}

\vfill

\end{frame}

$$\underbrace{A = \begin{pmatrix}-1&2\\-4&5\end{pmatrix}}_{\text{our running example}}, \qquad \det(tI - A) = (t-1)(t-3)$$

\vfill

Eigenvalues: $\lambda_1 = 1$, $\lambda_2 = 3$.

\vfill\pause

$$P_1 = \frac{A - 3I}{1 - 3} = \begin{pmatrix}2&-1\\2&-1\end{pmatrix}, \qquad

\x{Value tables} (from L15):

\end{tabular}

\vfill

$$A = \underbrace{1}_{\lambda_1} \cdot P_1 + \underbrace{3}_{\lambda_2} \cdot P_3$$

\end{frame}

\begin{frame}{What Does $P_i$ Do to a Vector?}

Pick $\mathbf{v} = \begin{pmatrix}2\\3\end{pmatrix}$. \quad Apply each projection:

\vfill

$$\bl{P_1 \mathbf{v}} = \begin{pmatrix}2&-1\\2&-1\end{pmatrix}\begin{pmatrix}2\\3\end{pmatrix} = \bl{\begin{pmatrix}1\\1\end{pmatrix}}$$

\vfill

$$\rd{P_3 \mathbf{v}} = \begin{pmatrix}-1&1\\-2&2\end{pmatrix}\begin{pmatrix}2\\3\end{pmatrix} = \rd{\begin{pmatrix}1\\2\end{pmatrix}}$$

\vfill\pause

Check: $\;\bl{\begin{pmatrix}1\\1\end{pmatrix}} + \rd{\begin{pmatrix}1\\2\end{pmatrix}} = \begin{pmatrix}2\\3\end{pmatrix} = \mathbf{v}\;\;\checkmark$

\end{frame}

% This is the audience's trusted tool from L15. We do NOT expand

\begin{frame}[shrink=15]{Why? --- Value Table Proof}

\x{Why} does $A$ scale the $P_1$-piece by $\lambda_1$?

\vfill

\x{Multiply} value tables entry-wise:

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|cc}
 & $t = 1$ & $t = 3$ \\
\hline
$A$ & $1$ & $3$ \\
$\times\; P_1$ & $1$ & $0$ \\
\hline
$AP_1$ & $\x{1}$ & $\x{0}$
\end{tabular}
\end{center}

\vfill\pause

Value table of $\lambda_1 P_1 = 1 \cdot P_1$:

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|cc}
 & $t = 1$ & $t = 3$ \\
\hline
$1 \cdot P_1$ & $\x{1}$ & $\x{0}$
\end{tabular}
\end{center}

\x{Same value table} $\implies$ \x{same matrix}: $\;\; AP_1 = \lambda_1 P_1 = 1 \cdot P_1$.

\vfill\pause

Similarly for $P_3$:

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|cc}
 & $t = 1$ & $t = 3$ \\
\hline
$A$ & $1$ & $3$ \\
$\times\; P_3$ & $0$ & $1$ \\
\hline
$AP_3$ & $\x{0}$ & $\x{3}$
\end{tabular}
$\;=\;$ value table of $3 P_3$ \;\checkmark
\end{center}

\end{frame}


\begin{frame}{$A$ Commutes With Each $P_i$}

What about $P_1 A$? \x{Multiply} value tables in the other order:

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|cc}
 & $t = 1$ & $t = 3$ \\
\hline
$P_1$ & $1$ & $0$ \\
$\times\; A$ & $1$ & $3$ \\
\hline
$P_1 A$ & $\x{1}$ & $\x{0}$
\end{tabular}
\end{center}

Same value table as $AP_1$! \quad So $P_1 A = AP_1 = \lambda_1 P_1$.

\vfill\pause

\begin{center}
\fbox{\parbox{0.85\textwidth}{\centering
Value table multiplication is \x{entry-wise}: order doesn't matter.\\[6pt]
$AP_i = P_i A = \lambda_i P_i \qquad\text{for each } i$
}}
\end{center}

\end{frame}


\begin{frame}[shrink=10]{Answer: $\operatorname{Col}(P_i)$ is the \x{Eigenspace}}

\begin{defi}[Eigenspace]
The \x{eigenspace} of $A$ for eigenvalue $\lambda_i$ is
$$E_{\lambda_i} \;=\; \bigl\{\,\mathbf{x} \in \mathbb{R}^n : A\mathbf{x} = \lambda_i \mathbf{x}\,\bigr\}$$
\end{defi}

\vfill\pause

If $\mathbf{x} \in \operatorname{Col}(P_i)$, then $P_i$ fixes it: $P_i\mathbf{x} = \mathbf{x}$.

$$A\mathbf{x} = A \underbrace{P_i \mathbf{x}}_{= \mathbf{x}} = \underbrace{AP_i}_{= \lambda_i P_i}\mathbf{x} = \lambda_i \underbrace{P_i\mathbf{x}}_{= \mathbf{x}} = \lambda_i \mathbf{x} \qquad\implies\quad \operatorname{Col}(P_i) \subseteq E_{\lambda_i}$$

\vfill\pause

Conversely: $P_i$ is a polynomial of $A$ (Lagrange formula). If $A\mathbf{x} = \lambda_i\mathbf{x}$, then $f(A)\mathbf{x} = f(\lambda_i)\mathbf{x}$ for any polynomial $f$. Since $P_i$ evaluates to $1$ at $\lambda_i$:

$$P_i\mathbf{x} = 1 \cdot \mathbf{x} = \mathbf{x} \qquad\implies\quad E_{\lambda_i} \subseteq \operatorname{Col}(P_i)$$

\vfill

$$\boxed{E_{\lambda_i} = \operatorname{Col}(P_i)}$$

\end{frame}


\begin{frame}{Check on Our Example}

$E_1 = \operatorname{Col}(P_1) = \operatorname{span}\left\{\begin{pmatrix}1\\1\end{pmatrix}\right\}$. \quad
$A\begin{pmatrix}1\\1\end{pmatrix} = \begin{pmatrix}1\\1\end{pmatrix} = \x{1}\begin{pmatrix}1\\1\end{pmatrix}$ \;\checkmark

\vfill

$E_3 = \operatorname{Col}(P_3) = \operatorname{span}\left\{\begin{pmatrix}1\\2\end{pmatrix}\right\}$. \quad
$A\begin{pmatrix}1\\2\end{pmatrix} = \begin{pmatrix}3\\6\end{pmatrix} = \x{3}\begin{pmatrix}1\\2\end{pmatrix}$ \;\checkmark

\vfill

\begin{center}
\fbox{\parbox{0.75\textwidth}{\centering
$P_i$ projects onto $E_{\lambda_i}$:\\[4pt]
the subspace where $A$ acts as \x{scalar multiplication by $\lambda_i$}.
}}
\end{center}

\end{frame}

\section{Compatible Projections Revisited}


\begin{frame}{What We Want to Show}

\vfill

\x{Two main points} in this section:

\vfill

\begin{enumerate}
\item Any vector $\mathbf{v}$ can be \x{decomposed into eigenvectors} --- one from each eigenspace $E_{\lambda_i}$.

\vspace{0.3cm}

\item Eigenvectors with \x{different eigenvalues} are \x{linearly independent}: the decomposition is unique.
\end{enumerate}

\vfill

The key: $P_1, \ldots, P_m$ are \x{compatible projections} (Lecture 8--9).

\vfill

\end{frame}


\begin{frame}[shrink=5]{Recall: Where Did $P_1, P_3$ Come From?}

\begin{cor}

\end{cor}

\begin{frame}[shrink=5]{Why Lagrange Does Not Apply Yet}

Lecture 15 worked because the annihilating polynomial had \x{distinct roots}:

$$h(t) = (t-\lambda_1)\cdots(t-\lambda_m), \qquad h(A)=0$$

\vfill

Then remainders have degree $<m$, so the remainder space has dimension $m$.

\vfill

\x{Here} the characteristic polynomial is

$$\det(tI - B) = (t-2)^2(t-5), \qquad (B-2I)^2(B-5I)=0.$$

\vfill

$$a + bt + ct^2 \qquad \text{a \x{3-dimensional} space.}$$

\vfill

\vfill

\begin{center}

\end{center}

$$q(t) = (t-2)(t-5).$$

\section{Diagonalization: Stacking the Cross-Fillings}

\begin{frame}[shrink=15]{From Sum to Product}

$$B \;=\; \rd{2}\cdot\rd{R_1} + \bl{2}\cdot\bl{R_2} + \gn{5}\cdot\gn{P_5}$$

\vfill

Write with \x{general variable $\lambda$} replacing the eigenvalues:

$$\rd{\lambda}\cdot\rd{\begin{pmatrix}1\\0\\0\end{pmatrix}}\rd{\begin{pmatrix}1&0&-1\end{pmatrix}} + \bl{\lambda}\cdot\bl{\begin{pmatrix}0\\1\\0\end{pmatrix}}\bl{\begin{pmatrix}0&1&-1\end{pmatrix}} + \gn{\lambda}\cdot\gn{\begin{pmatrix}1\\1\\1\end{pmatrix}}\gn{\begin{pmatrix}0&0&1\end{pmatrix}}$$

\vfill

$$= \underbrace{\begin{pmatrix}\rd{1}&\bl{0}&\gn{1}\\\rd{0}&\bl{1}&\gn{1}\\\rd{0}&\bl{0}&\gn{1}\end{pmatrix}}_{P} \underbrace{\begin{pmatrix}\rd{\lambda}&&\\&\bl{\lambda}&\\&&\gn{\lambda}\end{pmatrix}}_{D(\lambda)} \underbrace{\begin{pmatrix}\rd{1}&\rd{0}&\rd{-1}\\\bl{0}&\bl{1}&\bl{-1}\\\gn{0}&\gn{0}&\gn{1}\end{pmatrix}}_{Q}$$

\vfill

\x{Same $P$, same $Q$} --- only the \x{diagonal changes}.

\end{frame}


\begin{frame}[shrink=10]{The General Formula: $g(B) = P\,g(D)\,Q$}

For \x{any} function $g$, replace each diagonal entry $\lambda_i$ by $g(\lambda_i)$:

$$g(B) = P\begin{pmatrix}\rd{g(\lambda_1)}&&\\&\bl{g(\lambda_2)}&\\&&\gn{g(\lambda_3)}\end{pmatrix}Q$$

\vfill\pause

Set $g(\lambda) = 1$ for all $\lambda$:

$$\underbrace{\rd{1}\cdot\rd{R_1} + \bl{1}\cdot\bl{R_2} + \gn{1}\cdot\gn{P_5}}_{= P_2 + P_5 = I\;\text{(value table from \S2)}} = P\begin{pmatrix}\rd{1}&&\\&\bl{1}&\\&&\gn{1}\end{pmatrix}Q = PQ$$

\vfill

$$\boxed{PQ = I, \qquad\text{so}\quad Q = P^{-1}}$$

\vfill\pause

Set $g(\lambda) = \lambda$ (identity): $\quad B = PDP^{-1}$. \quad Set $g(\lambda) = \lambda^n$: $\quad B^n = PD^nP^{-1}$.

\end{frame}


\begin{frame}[shrink=5]{Diagonalization: $B = PDP^{-1}$}

$$\underbrace{\begin{pmatrix}2&0&3\\0&2&3\\0&0&5\end{pmatrix}}_{B} = \underbrace{\begin{pmatrix}\rd{1}&\bl{0}&\gn{1}\\\rd{0}&\bl{1}&\gn{1}\\\rd{0}&\bl{0}&\gn{1}\end{pmatrix}}_{P} \underbrace{\begin{pmatrix}\rd{2}&&\\&\bl{2}&\\&&\gn{5}\end{pmatrix}}_{D} \underbrace{\begin{pmatrix}\rd{1}&\rd{0}&\rd{-1}\\\bl{0}&\bl{1}&\bl{-1}\\\gn{0}&\gn{0}&\gn{1}\end{pmatrix}}_{P^{-1}}$$

\vfill

\x{Verify:}
$PD = \begin{pmatrix}2&0&5\\0&2&5\\0&0&5\end{pmatrix}$, \quad
$PDP^{-1} = \begin{pmatrix}2&0&3\\0&2&3\\0&0&5\end{pmatrix} = B$ \;\checkmark

\vfill\pause

\rd{Columns of $P$} = right eigenvectors. \quad \rd{Rows of $P^{-1}$} = left eigenvectors.

\vfill

\rd{Diagonal of $D$} = eigenvalues (with repetition).

\end{frame}


\begin{frame}{The Cancellation Trick}

Why does $g(B) = P\,g(D)\,P^{-1}$?

\vfill

$$B^2 = (\overbrace{PDP^{-1}}^{B})(\overbrace{PDP^{-1}}^{B}) = PD\underbrace{\cancel{P^{-1}P}}_{= I}DP^{-1} = PD^2P^{-1}$$

\vfill

The $P^{-1}P$ in the middle \x{cancels}!

\vfill\pause

\x{Powers:}

$$B^n = P D^n P^{-1}, \qquad D^n = \begin{pmatrix}2^n&&\\&2^n&\\&&5^n\end{pmatrix}$$

\vfill

Diagonal matrices are trivial to exponentiate.

\end{frame}

\section{Our Restriction: The Minimal Polynomial}


% ADR: The tension -- B from S3 had a repeated char poly root.
% AUDIENCE EMOTION: "wait, our assumption said distinct factors!"
% AUDIENCE ATTENTION: the contradiction grabs attention.
\begin{frame}[shrink=5]{Eigenvalues Are Roots of Any Annihilating Polynomial}

\begin{thm}
If $p(A) = 0$ and $\lambda$ is an eigenvalue of $A$, then $p(\lambda) = 0$.
\end{thm}

\vfill

\textit{Proof:} Let $A\mathbf{x} = \lambda\mathbf{x}$ with $\mathbf{x} \neq \mathbf{0}$. Then:

$$\mathbf{0} = p(A)\mathbf{x} = p(\lambda)\mathbf{x}$$

Since $\mathbf{x} \neq \mathbf{0}$: $\;\;p(\lambda) = 0$. \qed

\vfill\pause

\x{Consequence}: Every annihilating polynomial must have \x{all eigenvalues} as roots.

\vfill

Our spectral decomposition uses an annihilating polynomial with \x{distinct linear factors}. If the characteristic polynomial $\det(tI - A)$ has repeated factors, we need a \x{simpler} annihilating polynomial.

\end{frame}


\begin{frame}{A Repeated Root That Worked}

$B$ from \S3 had $\det(tI - B) = (t-2)^{\x{2}}(t-5)$ --- a \x{repeated} root!

\vfill

Yet the spectral machinery worked perfectly:
\begin{itemize}
\item Two distinct projections $P_2, P_5$ \;\checkmark
\item Compatible, partition of $I$ \;\checkmark
\item Cross-filling, eigenvectors \;\checkmark
\item $B = PDP^{-1}$ \;\checkmark
\end{itemize}

\vfill\pause

\x{Why?} There exists a simpler annihilating polynomial with distinct factors.

\end{frame}


\begin{frame}{The Minimal Polynomial}

\begin{defi}[Minimal Polynomial]
The \x{minimal polynomial} $m(t)$ is the lowest-degree monic polynomial such that $m(A) = 0$.
\end{defi}

\vfill

It divides every annihilating polynomial (including $\det(tI - A)$).

\vfill\pause

\x{Check on $B$:}

$$(B - 2I)(B - 5I) = \begin{pmatrix}0&0&3\\0&0&3\\0&0&3\end{pmatrix}\begin{pmatrix}-3&0&3\\0&-3&3\\0&0&0\end{pmatrix} = \begin{pmatrix}0&0&0\\0&0&0\\0&0&0\end{pmatrix}$$

\vfill

So $m(t) = (t-2)(t-5)$ --- \x{distinct} linear factors. \;\checkmark

\end{frame}


% ADR: A = 3I_2 example -- char poly (t-3)^2 but min poly (t-3).
% AUDIENCE EMOTION: "even a scalar matrix is fine"
% AUDIENCE ATTENTION: extreme case, grounding.
\begin{frame}{Another Fine Example: $A = 3I$}

$$A = \begin{pmatrix}3&0\\0&3\end{pmatrix}, \qquad \det(tI - A) = (t-3)^2 \quad\text{(repeated!)}$$

\vfill

But $A - 3I = 0$ already, so $m(t) = t - 3$ \quad (degree 1, one distinct factor).

\vfill

Spectral decomposition: $A = 3 \cdot I$. Just one projection $P_1 = I$.

\vfill

The whole space $\mathbb{R}^2$ is one eigenspace. \quad Fine!

\end{frame}

\vfill

$J - 2I = \begin{pmatrix}0&1\\0&0\end{pmatrix} \neq 0$. \qquad $(J - 2I)^2 = 0$.

\vfill

So $m(t) = (t-2)^2$ --- \x{repeated} factor!

\begin{frame}[shrink=5]{The Criterion for Diagonalizability}

\begin{thm}
$A$ is \x{diagonalizable} if and only if the minimal polynomial $m(t)$ factors into \x{distinct linear factors}:
$$m(t) = (t - \lambda_1)(t - \lambda_2)\cdots(t - \lambda_k), \qquad \lambda_i\text{ all distinct}$$
\end{thm}

\vfill\pause

($\Leftarrow$): Distinct linear factors $\implies$ spectral decomposition works $\implies$ diagonalizable. (This is exactly what we did in \S1--\S4.)

\vfill

($\Rightarrow$): If $A = PDP^{-1}$, then $(A - \lambda_1 I)\cdots(A - \lambda_k I) = P(D - \lambda_1 I)\cdots(D - \lambda_k I)P^{-1} = 0$, because every diagonal entry of $D$ is one of the $\lambda_i$'s.

\end{frame}


% ADR: Add the computational corollary requested by the instructor.
% The minimal polynomial criterion is theoretically clean but inconvenient:
% students can compute det(tI-A), but they usually cannot compute m(t)
% directly. The useful test is: factor the characteristic polynomial,
% delete repeated powers, then plug A into that square-free part.
% Terminology: "square-free part" is the standard computational phrase;
% "radical" is also used algebraically, but we introduce it only as a
% parenthetical synonym, not as new primary notation.
% AUDIENCE EMOTION: "this is a test I can actually perform."
\begin{frame}[shrink=5]{Practical Corollary: Delete the Repeated Powers}

\begin{cor}[Practical diagonalizability test]
A matrix $A$ is \x{diagonalizable} if and only if:

\begin{enumerate}
\item $\det(tI-A)$ factors into linear factors:
$$\det(tI-A)=(t-\lambda_1)^{a_1}\cdots(t-\lambda_k)^{a_k}.$$

\item After deleting the repeated powers, the remaining polynomial annihilates $A$:
$$\underbrace{(A-\lambda_1 I)\cdots(A-\lambda_k I)}_{\text{plug $A$ into the square-free part}}=0.$$
\end{enumerate}
\end{cor}

\vfill

The polynomial
$$q(t)=(t-\lambda_1)\cdots(t-\lambda_k)$$
is called the \x{square-free part} of $\det(tI-A)$.

\vfill

Some books also call it the \x{radical} of the characteristic polynomial.

\end{frame}


% ADR: Explain why the practical corollary is equivalent to the minimal
% polynomial theorem without asking students to compute the minimal
% polynomial. This is the bridge from theory to usable algorithm.
% AUDIENCE EMOTION: "the theorem became an algorithm."
\begin{frame}[shrink=5]{Why This Is the Same Criterion}

Let
$$q(t)=(t-\lambda_1)\cdots(t-\lambda_k)$$
be the square-free part of $\det(tI-A)$.

\vfill

\x{If $q(A)=0$:}

\smallskip

Then $q(t)$ is an annihilating polynomial with distinct linear factors.
So the minimal polynomial divides $q(t)$, hence also has distinct linear factors.

\vfill

By the theorem: $A$ is diagonalizable.

\vfill\pause

\x{If $A$ is diagonalizable:}

\smallskip

The minimal polynomial is exactly the product of the distinct eigenvalue factors:
$$m(t)=(t-\lambda_1)\cdots(t-\lambda_k)=q(t).$$

\vfill

So $q(A)=m(A)=0$.

\end{frame}


% ADR: Concrete examples immediately after the corollary. The user asked
% for several examples and a practical test. Each example follows the same
% three-line template: characteristic polynomial, delete powers, plug in.
% This repeated format is deliberate: students should leave with a recipe.
\begin{frame}[shrink=5]{Examples of the Practical Test}

\x{Example 1: repeated root, still diagonalizable}

$$B=\begin{pmatrix}2&0&3\\0&2&3\\0&0&5\end{pmatrix}, \qquad \det(tI-B)=(t-2)^2(t-5).$$

Delete powers: $q(t)=(t-2)(t-5)$.

$$(B-2I)(B-5I)=0 \qquad\implies\qquad B\text{ is diagonalizable.}$$

\vfill

\x{Example 2: scalar matrix}

$$A=3I_2, \qquad \det(tI-A)=(t-3)^2.$$

Delete powers: $q(t)=t-3$.

$$A-3I=0 \qquad\implies\qquad A\text{ is diagonalizable.}$$

\end{frame}


% ADR: Include a failure case right after the passing cases. The failure
% is essential because it shows the test is not just confirming what the
% characteristic polynomial already says. The exact same repeated-root
% characteristic shape can pass or fail depending on q(A).
\begin{frame}[shrink=5]{Example: Same Repeated Root, Different Result}

\x{Shear matrix}

$$J=\begin{pmatrix}2&1\\0&2\end{pmatrix}, \qquad \det(tI-J)=(t-2)^2.$$

\vfill

Delete powers: $q(t)=t-2$.

\vfill

Now test:

$$q(J)=J-2I=\begin{pmatrix}0&1\\0&0\end{pmatrix}\neq 0.$$

\vfill

So the square-free part does \x{not} annihilate $J$.

\vfill

$$\boxed{J\text{ is not diagonalizable.}}$$

\vfill

This is why repeated roots in $\det(tI-A)$ are not the issue. The issue is whether deleting the powers still gives an annihilating polynomial.

\end{frame}

% AUDIENCE EMOTION: "I can summarize this whole lecture in 4 lines"
% AUDIENCE ATTENTION: scannable, memorable.
\begin{frame}[shrink=5]{Summary}

\begin{enumerate}
\item $E_{\lambda_i} = \operatorname{Col}(P_i)$: eigenspace = column space of spectral projection.

\vspace{4pt}

\item Compatible projections (value tables verify instantly):
eigenvectors with different eigenvalues are independent; every vector decomposes uniquely.

\vspace{4pt}

\item Cross-fill $P_i = U_i V_i$: columns of $U_i$ = right eigenvectors,
rows of $V_i$ = left eigenvectors. $V_i U_j = \delta_{ij} I$.

\vspace{4pt}

\item Stack: $A = PDP^{-1}$, \; $g(A) = Pg(D)P^{-1}$ reduces every computation to plugging eigenvalues into $g$.

\vspace{4pt}

\item The \x{minimal polynomial} is the real restriction: distinct linear factors $\implies$ diagonalizable.
\end{enumerate}

\end{frame}

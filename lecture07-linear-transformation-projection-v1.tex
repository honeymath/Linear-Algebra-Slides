\section{Linear Transformations}

\begin{frame}{Definition}

\begin{defi}[Linear Transformation]
A function $T: V \to W$ between vector spaces is a \x{linear transformation} if
$$T(\lambda \mathbf{v} + \mu \mathbf{w}) = \lambda\, T(\mathbf{v}) + \mu\, T(\mathbf{w})$$
for all $\mathbf{v}, \mathbf{w} \in V$ and all scalars $\lambda, \mu$.
\end{defi}

\vspace{1em}

Equivalently, two separate properties:
\begin{enumerate}
\item \x{Additivity}: $T(\mathbf{v} + \mathbf{w}) = T(\mathbf{v}) + T(\mathbf{w})$
\item \x{Homogeneity}: $T(\lambda \mathbf{v}) = \lambda\, T(\mathbf{v})$
\end{enumerate}

\end{frame}

\begin{frame}{Linearity Extends to Any Linear Combination}

\begin{prop}[General Linearity]
If $T$ is a linear transformation, then for any vectors $\mathbf{v}_1, \ldots, \mathbf{v}_k$ and scalars $\lambda_1, \ldots, \lambda_k$:
$$T(\lambda_1 \mathbf{v}_1 + \cdots + \lambda_k \mathbf{v}_k) = \lambda_1 T(\mathbf{v}_1) + \cdots + \lambda_k T(\mathbf{v}_k)$$
\end{prop}

\textbf{Proof} (induction on $k$): Base case $k = 2$ is the definition.

\vspace{0.3em}

Inductive step: let $\mathbf{u} = \lambda_1 \mathbf{v}_1 + \cdots + \lambda_{k-1} \mathbf{v}_{k-1}$. Then:
\begin{align*}
T(\mathbf{u} + \lambda_k \mathbf{v}_k)
&= T(\mathbf{u}) + \lambda_k T(\mathbf{v}_k) & &\text{(definition)}\\
&= \lambda_1 T(\mathbf{v}_1) + \cdots + \lambda_k T(\mathbf{v}_k) & &\text{(induction hypothesis)} \quad\qed
\end{align*}

\end{frame}

\begin{frame}{Common Examples in $\mathbb{R}^2$}

\begin{enumerate}
\item \x{Rotation by $90^\circ$}:
$T\begin{pmatrix} x \\ y \end{pmatrix} = \begin{pmatrix} -y \\ x \end{pmatrix}$
\quad $A = \begin{pmatrix} 0 & -1 \\ 1 & 0 \end{pmatrix}$

\vspace{0.5em}

\item \x{Reflection across $x$-axis}:
$T\begin{pmatrix} x \\ y \end{pmatrix} = \begin{pmatrix} x \\ -y \end{pmatrix}$
\quad $A = \begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}$

\end{enumerate}

\end{frame}

\begin{frame}{Common Examples in $\mathbb{R}^2$ (continued)}

\begin{enumerate}
\setcounter{enumi}{2}

\item \x{Scaling by 2}:
$T\begin{pmatrix} x \\ y \end{pmatrix} = \begin{pmatrix} 2x \\ 2y \end{pmatrix}$
\quad $A = \begin{pmatrix} 2 & 0 \\ 0 & 2 \end{pmatrix}$

\vspace{1em}

\item \x{Projection onto $x$-axis}:
$T\begin{pmatrix} x \\ y \end{pmatrix} = \begin{pmatrix} x \\ 0 \end{pmatrix}$
\quad $A = \begin{pmatrix} 1 & 0 \\ 0 & 0 \end{pmatrix}$
\end{enumerate}

\vspace{1em}

In each case: the matrix $A$ \x{is} the linear transformation.

\end{frame}

\begin{frame}{Matrix Representation}

\begin{prop}[Matrix Representation]
Every linear transformation $T: \mathbb{R}^n \to \mathbb{R}^m$ can be written as
$$T(\mathbf{v}) = A\mathbf{v}$$
for some $m \times n$ matrix $A$.

Conversely, every matrix defines a linear transformation.
\end{prop}

\vspace{1em}

In other words: \x{matrices = linear transformations}. They are the same thing!

\end{frame}

\begin{frame}{Linear Operators}

\begin{defi}[Linear Operator]
When the domain and codomain are the same ($T: V \to V$), we call $T$ a \x{linear operator}.

Represented by \x{square matrices}.
\end{defi}

\vspace{1em}

For the rest of this lecture, we focus on \x{linear operators} (square matrices).

\end{frame}

\begin{frame}{Describing Projection Mathematically}

Now let us \x{mathematically describe} what a projection should do.

\vspace{0.5em}

Replace the floor by a subspace $W$, and the sunlight direction by a subspace $U$.

\vspace{0.5em}

Given a vector $\mathbf{v}$, we observe that its shadow $\mathbf{w}$ must satisfy:

\vspace{0.3em}

\begin{enumerate}
\item The shadow lands on the floor.

Mathematically: $\mathbf{w} \in W$

\vspace{0.3em}

\item The line from $\mathbf{v}$ to its shadow is parallel to the sunlight.

Mathematically: $\mathbf{v} - \mathbf{w} \in U$

\vspace{0.3em}

\item Every vector has a \x{unique} shadow.

Mathematically: such $\mathbf{w}$ exists and is unique for each $\mathbf{v}$
\end{enumerate}

\vspace{0.5em}

% (See ADR on "Describing Projection Mathematically" above for why no \begin{defi} here.)
These three conditions completely determine the map $\mathbf{v} \mapsto \mathbf{w}$.

\end{frame}

\begin{frame}{The Proof}

\begin{prop}[Projection gives the shadow]
If $P^2 = P$ and $\mathbf{w}$ is the geometric shadow of $\mathbf v$ (i.e.\ $\mathbf{w} \in \operatorname{Col}(P)$ and $\mathbf{v} - \mathbf{w} \in \operatorname{Null}(P)$), then $\mathbf{w} = P\mathbf{v}$.
\end{prop}

\textbf{Proof}:

Since $\mathbf{w} \in \operatorname{Col}(P)$, we have $\mathbf{w} = P\mathbf{x}$ for some $\mathbf{x}$.
$$P\mathbf{w} = P(P\mathbf{x}) = P^2\mathbf{x} = P\mathbf{x} = \mathbf{w}$$

Since $\mathbf{v} - \mathbf{w} \in \operatorname{Null}(P)$:
$$P(\mathbf{v} - \mathbf{w}) = \mathbf{0} \quad\implies\quad P\mathbf{v} = P\mathbf{w}$$

Combining:
$$P\mathbf{v} = P\mathbf{w} = \mathbf{w} \qquad \qed$$

\end{frame}

\section{Basic Properties of Projections}

\begin{frame}{Column Space Membership Test}

\begin{prop}[Floor Membership Test]
Let $P$ be a projection. For any $\mathbf{y} \in \mathbb{R}^n$:
$$\mathbf{y} \in \operatorname{Col}(P) \quad\iff\quad P\mathbf{y} = \mathbf{y}$$
\end{prop}

\textbf{Proof}:

($\Rightarrow$): If $\mathbf{y} \in \operatorname{Col}(P)$, then $\mathbf{y} = P\mathbf{x}$ for some $\mathbf{x}$.
$$P\mathbf{y} = P(P\mathbf{x}) = P^2\mathbf{x} = P\mathbf{x} = \mathbf{y} \quad\checkmark$$

($\Leftarrow$): If $P\mathbf{y} = \mathbf{y}$, then $\mathbf{y} \in \operatorname{Col}(P)$ by definition. $\quad\checkmark$

\end{frame}

\begin{frame}{Geometric Meaning: On the Floor $\iff$ Own Shadow}

\x{Geometric meaning}: A vector is on the floor $\iff$ it equals its own shadow.

\vspace{0.5em}

A vector $\mathbf{y}$ lies on the floor means its shadow coincides with itself:
$$\mathbf{y} \in \operatorname{Col}(P) \quad\iff\quad P\mathbf{y} = \mathbf{y}$$

\end{frame}

\begin{frame}{The Companion Projection: $I - P$}

\begin{prop}[$I - P$ is also a Projection]
If $P^2 = P$, then $I - P$ is also a projection.
\end{prop}

\textbf{Proof}:
$$(I - P)^2 = I - 2P + P^2 = I - 2P + P = I - P \quad\checkmark$$

\vspace{1em}

\x{Geometric interpretation}:

\begin{center}
\begin{tabular}{ll}
$P$ & projects \x{onto the floor} \\
$I - P$ & projects \x{in the direction of sunlight} \\
\end{tabular}
\end{center}

\vspace{0.5em}

For any $\mathbf{v}$:
$$\underbrace{P\mathbf{v}}_{\text{shadow on floor}} + \underbrace{(I-P)\mathbf{v}}_{\text{displacement toward sun}} = \mathbf{v}$$

\end{frame}

\begin{frame}{The Interchanging Property}

\begin{prop}[Column Space and Null Space Interchange]
For any projection $P$:
$$\operatorname{Col}(P) = \operatorname{Null}(I - P) \qquad\text{and}\qquad \operatorname{Col}(I - P) = \operatorname{Null}(P)$$
\end{prop}

\vspace{0.5em}

\begin{center}
\begin{tabular}{c|c}
\textbf{For $P$} & \textbf{For $I - P$} \\
\hline
$\operatorname{Col}(P)$ & $= \operatorname{Null}(I - P)$ \\
$\operatorname{Null}(P)$ & $= \operatorname{Col}(I - P)$ \\
\end{tabular}
\end{center}

\vspace{0.5em}

$P$ and $I - P$ \x{interchange} their column spaces and null spaces.

\end{frame}

\begin{frame}{Proof of the Interchanging Property}

\x{Show}: $\operatorname{Col}(P) = \operatorname{Null}(I - P)$.

\vspace{0.5em}

($\subseteq$): Take $\mathbf{v} \in \operatorname{Col}(P)$, so $P\mathbf{v} = \mathbf{v}$.

$\quad (I-P)\mathbf{v} = \mathbf{v} - P\mathbf{v} = \mathbf{v} - \mathbf{v} = \mathbf{0} \quad\implies\quad \mathbf{v} \in \operatorname{Null}(I-P) \;\checkmark$

\vspace{0.5em}

($\supseteq$): Take $\mathbf{v} \in \operatorname{Null}(I-P)$, so $(I-P)\mathbf{v} = \mathbf{0}$.

$\quad \mathbf{v} - P\mathbf{v} = \mathbf{0} \quad\implies\quad \mathbf{v} = P\mathbf{v} \quad\implies\quad \mathbf{v} \in \operatorname{Col}(P) \;\checkmark$

\vspace{0.5em}

The second equality $\operatorname{Col}(I-P) = \operatorname{Null}(P)$ follows by replacing $P$ with $I-P$.

\end{frame}

\begin{frame}{The Disjoint Property}

\begin{prop}[Column Space and Null Space Are Disjoint]
For any projection $P$:
$$\operatorname{Col}(P) \cap \operatorname{Null}(P) = \{\mathbf{0}\}$$
\end{prop}

\textbf{Proof}:

Let $\mathbf{v} \in \operatorname{Col}(P) \cap \operatorname{Null}(P)$.

\vspace{0.3em}

$\mathbf{v} \in \operatorname{Col}(P) \implies P\mathbf{v} = \mathbf{v}$

$\mathbf{v} \in \operatorname{Null}(P) \implies P\mathbf{v} = \mathbf{0}$

\vspace{0.3em}

Therefore $\mathbf{v} = \mathbf{0}$. \qed

\vspace{0.5em}

\x{Geometric meaning}: The floor and the sunlight direction \x{only meet at the origin}. A non-zero vector cannot simultaneously be on the floor and point toward the sun.

\end{frame}

% ADR: Reveal — the algebraic definition. Same pattern as §2: drop the answer

% AUDIENCE EMOTION: satisfaction — the pattern they spotted is now a theorem.
\begin{frame}{Algebraic Definition}

\begin{defi}[Orthogonal Projection]
A matrix $P$ is called an \x{orthogonal projection} if
$$P^2 = P \quad\text{and}\quad P = P^T$$
\end{defi}

\vspace{1em}

That's it: \x{idempotent + symmetric}.

\vspace{1em}

But why does matrix symmetry capture perpendicular sunlight?

\end{frame}

% ADR: The characterization theorem is the JUSTIFICATION, not the definition.
% It says: the algebraic condition (P = P^T) is equivalent to the geometric
% condition (Col ⊥ Null). This validates the rapport: the physical picture
% was not just a metaphor — it is a theorem. The audience's gut feeling
% ("the symmetric one looks more natural") is mathematically correct.
\begin{frame}{Why Symmetry = Perpendicular Sunlight}

\begin{thm}[Characterization of Orthogonal Projections]
Let $P$ be a projection ($P^2 = P$). The following are equivalent:
\begin{enumerate}
\item $P = P^T$ \quad (matrix symmetry)
\item $\operatorname{Col}(P) \perp \operatorname{Null}(P)$ \quad (sunlight $\perp$ floor)
\end{enumerate}
\end{thm}

\vspace{1em}

\x{Why this is powerful}: to check whether sunlight is perpendicular, just compare entries: $p_{ij} = p_{ji}$.

\vspace{0.5em}

No need to compute column spaces, null spaces, or inner products!

\end{frame}

\begin{frame}{Proof: ($P = P^T$) $\Rightarrow$ ($\operatorname{Col}(P) \perp \operatorname{Null}(P)$)}

Assume $P = P^T$. Take $\mathbf{v} \in \operatorname{Col}(P)$ and $\mathbf{w} \in \operatorname{Null}(P)$.

\vspace{0.5em}

By Lecture 6: $\operatorname{Col}(A^T) \perp \operatorname{Null}(A)$ always holds.

\vspace{0.5em}

Since $P^T = P$:
$$\operatorname{Col}(P) = \operatorname{Col}(P^T) \perp \operatorname{Null}(P) \quad\checkmark$$

\vspace{1em}

\x{Alternative direct proof}: Since $P\mathbf{v} = \mathbf{v}$:

$$\mathbf{v}^T\mathbf{w} = (P\mathbf{v})^T\mathbf{w} = \mathbf{v}^T \underbrace{P^T}_{= P}\mathbf{w} = \mathbf{v}^T \underbrace{P\mathbf{w}}_{= \mathbf{0}} = 0 \quad\checkmark$$

\end{frame}

\begin{frame}{Proof: ($\operatorname{Col}(P) \perp \operatorname{Null}(P)$) $\Rightarrow$ ($P = P^T$)}

Assume $\operatorname{Col}(P) \perp \operatorname{Null}(P)$. Take \x{any} vectors $\mathbf{v}, \mathbf{w}$.

\vspace{0.5em}

\begin{itemize}
\item $P\mathbf{v} \in \operatorname{Col}(P)$ \quad (output of $P$ is always in $\operatorname{Col}(P)$)
\item $(I - P)\mathbf{w} \in \operatorname{Null}(P)$ \quad (by interchanging property: $\operatorname{Col}(I-P) = \operatorname{Null}(P)$)
\end{itemize}

\vspace{0.5em}

By orthogonality, $(P\mathbf{v})^T (I - P)\mathbf{w} = 0$ for all $\mathbf{v}, \mathbf{w}$.

Now expand using the \x{transpose undressing rule} $(AB)^T = B^T A^T$:
\begin{align*}
0 &= (P\mathbf{v})^T (I - P)\mathbf{w} & &\text{(orthogonality)}\\
  &= \mathbf{v}^T P^T (I - P) \mathbf{w} & &\text{(undressing: $(P\mathbf{v})^T = \mathbf{v}^T P^T$)}\\
  &= \mathbf{v}^T (P^T - P^T P) \mathbf{w} & &\text{(distribute)}
\end{align*}

Since this holds for \x{all} $\mathbf{v}, \mathbf{w}$: $\; P^T - P^T P = 0$, so $P^T = P^T P$.

\end{frame}

\begin{frame}{Proof (continued): Concluding $P = P^T$}

We showed: $P^T = P^T P$. Now take the \x{transpose of both sides}:

\begin{align*}
(P^T)^T &= (P^T P)^T & &\text{(transpose both sides)} \\
P &= P^T \underbrace{(P^T)^T}_{= P} & &\text{(undressing + $(A^T)^T = A$)} \\
P &= P^T P & &
\end{align*}

Combining: $\;P^T = P^T P = P$, therefore $P = P^T$. \qed

\vspace{0.5em}

\x{Key principle}: If $\mathbf{v}^T A \mathbf{w} = \mathbf{v}^T B \mathbf{w}$ for all $\mathbf{v}, \mathbf{w}$, then $A = B$.

\end{frame}

\begin{frame}{Orthogonal Projection: Summary}

\begin{alertblock}{Orthogonal Projection = Two Algebraic Conditions}
$$\boxed{P^2 = P \quad\text{and}\quad P = P^T}$$

The geometry of ``sunlight perpendicular to floor'' is captured by \x{matrix symmetry}.
\end{alertblock}

\end{exampleblock}

\end{frame}
